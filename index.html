Google Apps Script Setup Instructions

  1. Create a new Google Sheet in your Google Drive.
  2. Go to Extensions > Apps Script.
  3. Paste the snippet below into Code.gs and click Save:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Auto-create header row if sheet is empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp", "Student Name", "Teammates", 
        "Q3 Contributions", "Q4 Specific Part", "Q5 Decision", 
        "Q6 Prep Work", "Q7 Self %", "Q8 Evidence", 
        "Q9 Group Allocations", "Q10 Peer Descriptions", 
        "Q11 Additional Context"
      ]);
    }
    
    sheet.appendRow([
      new Date(),
      data.studentName || "",
      (data.teammates || []).join(", "),
      data.q3_contributions || "",
      data.q4_specificPart || "",
      data.q5_decision || "",
      data.q6_prep || "",
      data.q7_selfPercent + "%",
      data.q8_evidence || "",
      JSON.stringify(data.q9_allocations || {}),
      JSON.stringify(data.q10_peerDescriptions || {}),
      data.q11_distributionContext || ""
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"result":"success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"result":"error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
  1. Click Deploy > New deployment.
  2. Select type: Web App.
  3. Set Execute as: "Me" and Who has access: "Anyone".
  4. Deploy, copy the Web App URL (ends in /exec), and replace PLACE_GOOGLE_URL_HERE in your website code or enter it in the "Sheet Config" button!
